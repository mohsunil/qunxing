# Country scope. Create AND APPLY a proles-free template; modify_species alone
# creates an unused template in 4.5.2. Exact matching preserves gene variants.
haipam_repair_gray_species = {
 every_owned_species = {
  limit = { has_trait = trait_haipam_gray has_trait = trait_syncretic_proles }
  save_event_target_as = haipam_gray_old
  modify_species = {
   species = this
   remove_trait = trait_syncretic_proles
   effect = {
    save_event_target_as = haipam_gray_repaired
   }
  }
  event_target:haipam_gray_repaired = { set_citizenship_type = { country = root type = citizenship_limited } }
  root = {
   every_owned_planet = {
    every_owned_pop_group = {
     limit = { is_exact_same_species = event_target:haipam_gray_old }
     change_species = event_target:haipam_gray_repaired
    }
   }
  }
 }
}

haipam_reset_pipeline = {
 set_variable = { which = hp_worker_0 value = 0 }
 set_variable = { which = hp_worker_1 value = 0 }
 set_variable = { which = hp_worker_2 value = 0 }
 set_variable = { which = hp_worker_3 value = 0 }
 set_variable = { which = hp_specialist_0 value = 0 }
 set_variable = { which = hp_specialist_1 value = 0 }
 set_variable = { which = hp_specialist_2 value = 0 }
 set_variable = { which = hp_specialist_3 value = 0 }
}

haipam_tick_worker = {
 export_trigger_value_to_variable = { trigger = free_jobs_of_type parameters = { category = worker } variable = hp_free }
 set_variable = { which = hp_birth value = hp_worker_3 }
 if = { limit = { check_variable = { which = hp_birth value > hp_free } } set_variable = { which = hp_birth value = hp_free } }
 if = {
  limit = { check_variable = { which = hp_birth value > 0 } }
  create_pop_group = { species = event_target:haipam_hatch_species category = worker size = hp_birth }
  set_variable = { which = hp_sub value = hp_birth }
  multiply_variable = { which = hp_sub value = -1 }
  change_variable = { which = hp_free value = hp_sub }
 }
 set_variable = { which = hp_worker_3 value = hp_worker_2 }
 set_variable = { which = hp_worker_2 value = hp_worker_1 }
 set_variable = { which = hp_worker_1 value = hp_worker_0 }
 set_variable = { which = hp_worker_0 value = 0 }

 if = { limit = { check_variable = { which = hp_worker_3 value > hp_free } } set_variable = { which = hp_worker_3 value = hp_free } }
 set_variable = { which = hp_sub value = hp_worker_3 }
 multiply_variable = { which = hp_sub value = -1 }
 change_variable = { which = hp_free value = hp_sub }

 if = { limit = { check_variable = { which = hp_worker_2 value > hp_free } } set_variable = { which = hp_worker_2 value = hp_free } }
 set_variable = { which = hp_sub value = hp_worker_2 }

 change_variable = { which = hp_free value = hp_sub }

 if = { limit = { check_variable = { which = hp_worker_1 value > hp_free } } set_variable = { which = hp_worker_1 value = hp_free } }
 set_variable = { which = hp_sub value = hp_worker_1 }
 multiply_variable = { which = hp_sub value = -1 }
