   add_building = building_haipam_royal_hatchery

  log = "HAIPAM_INIT_COMPLETE"

}
country_event = {
 id = haipam.10 hide_window = yes is_triggered_only = yes
 trigger = { haipam_is_empire = yes has_country_flag = haipam_initialized any_owned_species = { haipam_is_gray = yes } }
 immediate = {
  # Also repairs already-started alpha saves after their next monthly pulse.
  haipam_repair_gray_species = yes
  haipam_refresh_visuals = yes
  every_owned_leader = { limit = { species = { NOT = { has_trait = trait_haipam_white } } } change_species = owner_main_species }
  haipam_refresh_role_portraits = yes
  every_owned_planet = {
   if = {
    limit = { NOT = { has_planet_flag = haipam_pipeline_initialized } }
    haipam_reset_pipeline = yes
    set_planet_flag = haipam_pipeline_initialized
   }
   if = {
    limit = { has_active_building = building_haipam_royal_hatchery }
    # Prefer a local gray template; fall back to an owned gray species.
    if = {
     limit = { any_owned_pop_group = { haipam_is_gray_pop_group = yes } }
     random_owned_pop_group = { limit = { haipam_is_gray_pop_group = yes } species = { save_event_target_as = haipam_hatch_species } }
    }
    else = { owner = { random_owned_species = { limit = { haipam_is_gray = yes } save_event_target_as = haipam_hatch_species } } }
    if = {
     limit = { exists = event_target:haipam_hatch_species }
     haipam_tick_worker = yes
     haipam_tick_specialist = yes
    }
   }
   else = { haipam_reset_pipeline = yes }
   # Give employment allocation time; never delete employed gray groups.
   colony_event = { id = haipam.20 days = 15 }
  }
 }
}
colony_event = {
 id = haipam.20 hide_window = yes is_triggered_only = yes
 trigger = { exists = owner owner = { haipam_is_empire = yes } }
 immediate = {
  every_owned_pop_group = {
   limit = { haipam_is_gray_pop_group = yes OR = { is_unemployed = yes is_pop_category = civilian } }
   # Same explicit deletion interface used by native virtual-pop cleanup.
   log = "HAIPAM_SURPLUS_GRAY_GROUP_REMOVED"
   kill_pop_group = { pop_group = this percentage = 1 }
  }
 }
}
